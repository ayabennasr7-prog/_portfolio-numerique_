🌸 Projet 1 : Calcul de la somme

Ce programme calcule la somme de deux nombres.

#include <stdio.h>

int main() {
    int a, b;
    printf("Donner deux nombres : ");
    scanf("%d %d", &a, &b);

    printf("La somme est : %d", a + b);
    return 0;
}

🔢 Projet 2 : Pair ou impair

Ce programme détermine si un nombre est pair ou impair.

#include <stdio.h>

int main() {
    int n;
    printf("Donner un nombre : ");
    scanf("%d", &n);

    if (n % 2 == 0)
        printf("Nombre pair");
    else
        printf("Nombre impair");

    return 0;
}

✨ Projet 3 : Factorielle

Ce programme calcule la factorielle d'un entier positif.

#include <stdio.h>

int main() {
    int n, i;
    long long fact = 1;

    printf("Donner un entier positif : ");
    scanf("%d", &n);

    if (n < 0) {
        printf("Nombre invalide");
    } else {
        for (i = 1; i <= n; i++) {
            fact = fact * i;
        }
        printf("Factorielle = %lld", fact);
    }

    return 0;
}
